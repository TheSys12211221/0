git init - создает новый локальный репозиторий
git clone <url> - копирует удаленный репозиторий на компьютер
git status - показывает состояние файлов и индекса
git add <файл> - добавляет изменения в индекс для коммита
git commit -m "сообщение" - фиксирует изменения в истории
git push - отправляет коммиты в удаленный репозиторий
git pull - загружает и объединяет изменения из удаленного репозитория
git log - показывает историю коммитов
git branch - показывает список веток
git checkout - переключает на указанную ветку
git merge - вливает указанную ветку в текущую
